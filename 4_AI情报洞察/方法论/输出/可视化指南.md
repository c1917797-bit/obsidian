## 三、图1: 42格热力图

### 3.1 用途
展示 7 对象 × 6 方法的论文密度/热度，用于 1.2 技术全景图。

### 3.2 代码

```python
# -*- coding: utf-8 -*-
"""chart_01_heatmap.py - 42格热力图"""
import sys
import json
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np
from collections import defaultdict
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

OBJECTS = ['W(权重)', 'A(激活)', 'K(KV)', 'C(通信)', 'M(模态)', 'V(声码器)', 'S(状态)']
METHODS = ['Q(量化)', 'L(低秩)', 'S(稀疏)', 'P(剪枝)', 'D(蒸馏)', 'G(通用)']
GRID_KEYS = [
    ['W×Q','W×L','W×S','W×P','W×D','W×G'],
    ['A×Q','A×L','A×S','A×P','A×D','A×G'],
    ['K×Q','K×L','K×S','K×P','K×D','K×G'],
    ['C×Q','C×L','C×S','C×P','C×D','C×G'],
    ['M×Q','M×L','M×S','M×P','M×D','M×G'],
    ['V×Q','V×L','V×S','V×P','V×D','V×G'],
    ['S×Q','S×L','S×S','S×P','S×D','S×G'],
]

def load_grid_counts(meta_path):
    """从 papers_meta_v2.json 加载格子计数"""
    with open(meta_path, 'r', encoding='utf-8') as f:
        papers = json.load(f)
    counts = defaultdict(int)
    for p in papers:
        g = p.get('grid')
        if g and g != 'Unclassified':
            counts[g] += 1
    return counts

def draw_heatmap(counts, output_path):
    """绘制 42 格热力图"""
    data = np.zeros((7, 6))
    labels = [['' for _ in range(6)] for _ in range(7)]
    
    for i in range(7):
        for j in range(6):
            key = GRID_KEYS[i][j]
            data[i][j] = counts.get(key, 0)
            labels[i][j] = f"{key}\n{int(data[i][j])}"
    
    fig, ax = plt.subplots(figsize=(10, 8))
    
    im = ax.imshow(data, cmap='YlOrRd', aspect='auto', vmin=0, vmax=max(1, data.max()))
    
    ax.set_xticks(np.arange(6))
    ax.set_yticks(np.arange(7))
    ax.set_xticklabels(METHODS, fontsize=12)
    ax.set_yticklabels(OBJECTS, fontsize=12)
    
    for i in range(7):
        for j in range(6):
            color = 'white' if data[i][j] > data.max() * 0.6 else 'black'
            ax.text(j, i, labels[i][j], ha='center', va='center',
                    color=color, fontsize=9, fontweight='bold')
    
    ax.set_title('推理压缩 42格技术热力图（论文数量）', fontsize=16, pad=20)
    
    cbar = plt.colorbar(im, ax=ax, shrink=0.8)
    cbar.set_label('论文数量', fontsize=12)
    
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close()
    print(f"输出: {output_path}")

if __name__ == '__main__':
    META_PATH = r'C:\Users\Huawei\Documents\code\Obsidian\4_AI情报洞察\论文洞察\papers_meta_v2.json'
    OUTPUT = r'C:\Users\Huawei\Documents\code\Obsidian\4_AI情报洞察\方法论\测评\chart_01_heatmap.png'
    
    counts = load_grid_counts(META_PATH)
    print("格子计数:")
    for g, c in sorted(counts.items(), key=lambda x: -x[1]):
        print(f"  {g}: {c}")
    draw_heatmap(counts, OUTPUT)
```

```python
# -*- coding: utf-8 -*-
"""chart_02_trend.py - 月度趋势曲线"""
import sys
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from collections import defaultdict
from datetime import datetime

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

def draw_trend(month_data, title, output_path, target_grids=None):
    """
    month_data: {month_str: {grid: count}}
    target_grids: 要绘制的格子列表
    """
    if target_grids is None:
        target_grids = ['W×Q', 'K×Q', 'S×G', 'C×G']
    
    months = sorted(month_data.keys())
    
    fig, ax = plt.subplots(figsize=(12, 6))
    
    colors = ['#2563EB', '#DC2626', '#10B981', '#F59E0B']
    markers = ['o', 's', '^', 'D']
    
    for idx, grid in enumerate(target_grids):
        values = [month_data[m].get(grid, 0) for m in months]
        ax.plot(months, values, marker=markers[idx % len(markers)],
                color=colors[idx % len(colors)], linewidth=2, markersize=8,
                label=grid)
    
    ax.set_xlabel('月份', fontsize=12)
    ax.set_ylabel('新增论文数', fontsize=12)
    ax.set_title(title, fontsize=16, pad=15)
    ax.legend(loc='upper left', fontsize=11)
    ax.grid(True, alpha=0.3, linestyle='--')
    
    ax.axhline(y=5, color='gray', linestyle=':', alpha=0.5, label='爆发阈值(5)')
    
    plt.xticks(rotation=45)
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close()
    print(f"输出: {output_path}")
```

```python
# -*- coding: utf-8 -*-
"""chart_03_priority.py - 优先级矩阵（四象限）"""
import sys
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

def draw_priority_matrix(projects, output_path):
    """
    projects: list of {name, score, risk, priority}
    score: 综合得分 (0-12)
    risk: 反方证据 (0-15)
    """
    fig, ax = plt.subplots(figsize=(10, 8))
    
    # 四象限分割线
    ax.axhline(y=6, color='gray', linestyle='--', alpha=0.5)
    ax.axvline(x=8, color='gray', linestyle='--', alpha=0.5)
    
    # 四象限标签
    ax.text(10, 13, '⭐ 明星区\n(高分低风险)', ha='center', fontsize=12,
            color='#10B981', fontweight='bold')
    ax.text(5, 13, '⚠️ 赌注区\n(高分高风险)', ha='center', fontsize=12,
            color='#F59E0B', fontweight='bold')
    ax.text(10, 3, '📝 备选区\n(低分低风险)', ha='center', fontsize=12,
            color='#6B7280', fontweight='bold')
    ax.text(5, 3, '❌ 淘汰区\n(低分高风险)', ha='center', fontsize=12,
            color='#DC2626', fontweight='bold')
    
    color_map = {'P0': '#DC2626', 'P1': '#F59E0B', 'P2': '#10B981'}
    
    for p in projects:
        ax.scatter(p['risk'], p['score'], s=200,
                   color=color_map.get(p['priority'], '#6B7280'),
                   edgecolors='black', linewidth=1.5, zorder=5)
        ax.annotate(p['name'], (p['risk'], p['score']),
                    xytext=(8, 8), textcoords='offset points',
                    fontsize=10, fontweight='bold')
    
    ax.set_xlabel('反方证据得分（风险 →）', fontsize=13)
    ax.set_ylabel('综合得分（价值 ↑）', fontsize=13)
    ax.set_title('课题优先级矩阵', fontsize=16, pad=15)
    ax.set_xlim(0, 15)
    ax.set_ylim(0, 14)
    ax.grid(True, alpha=0.2)
    
    # 图例
    from matplotlib.patches import Patch
    legend_elements = [
        Patch(facecolor='#DC2626', label='P0 立项'),
        Patch(facecolor='#F59E0B', label='P1 季度内'),
        Patch(facecolor='#10B981', label='P2 观察'),
    ]
    ax.legend(handles=legend_elements, loc='lower right', fontsize=11)
    
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close()
    print(f"输出: {output_path}")
```

### 6.2 代码

```python
# -*- coding: utf-8 -*-
"""chart_05_radar.py - 风险雷达图（反方证据5维）"""
import sys
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

def draw_radar(projects, output_path):
    """
    projects: list of {name, scores: {dim: value}}
    dims: 历史失败, 专利壁垒, 资源门槛, 团队匹配, 反向场景
    values: 1-3
    """
    categories = ['历史失败', '专利壁垒', '资源门槛', '团队匹配', '反向场景']
    N = len(categories)
    
    angles = np.linspace(0, 2 * np.pi, N, endpoint=False).tolist()
    angles += angles[:1]
    
    fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(polar=True))
    
    colors = ['#DC2626', '#F59E0B', '#10B981', '#2563EB']
    
    for idx, p in enumerate(projects):
        values = [p['scores'].get(c, 1) for c in categories]
        values += values[:1]
        ax.plot(angles, values, 'o-', linewidth=2,
                color=colors[idx % len(colors)], label=p['name'], markersize=6)
        ax.fill(angles, values, alpha=0.15, color=colors[idx % len(colors)])
    
    ax.set_xticks(angles[:-1])
    ax.set_xticklabels(categories, fontsize=12)
    ax.set_ylim(0, 3)
    ax.set_yticks([1, 2, 3])
    ax.set_yticklabels(['1\n低', '2\n中', '3\n高'], fontsize=10)
    
    ax.set_title('反方证据风险雷达图', fontsize=16, pad=20)
    ax.legend(loc='upper right', bbox_to_anchor=(1.3, 1.1), fontsize=11)
    
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close()
    print(f"输出: {output_path}")

if __name__ == '__main__':
    sample = [
```

```python
# -*- coding: utf-8 -*-
"""chart_04_gantt.py - 课题甘特图"""
import sys
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

def draw_gantt(tasks, output_path):
    """
    tasks: list of {name, start_week, duration_weeks, priority, phase}
    phase: '验证', '实施', '收尾'
    """
    fig, ax = plt.subplots(figsize=(14, 5))
    
    phase_colors = {
        '验证': '#3B82F6',
        '实施': '#10B981',
        '收尾': '#F59E0B',
    }
    
    y_pos = len(tasks)
    
    for task in tasks:
        color = phase_colors.get(task['phase'], '#6B7280')
        ax.barh(y_pos, task['duration_weeks'], left=task['start_week'],
                height=0.6, color=color, edgecolor='black', linewidth=0.5)
        ax.text(task['start_week'] + task['duration_weeks'] / 2, y_pos,
                task['name'], ha='center', va='center', fontsize=10,
                fontweight='bold', color='white' if task['duration_weeks'] > 3 else 'black')
        y_pos -= 1
    
    ax.set_xlabel('周次（Week）', fontsize=12)
    ax.set_ylabel('课题', fontsize=12)
    ax.set_title('课题时间线（甘特图）', fontsize=16, pad=15)
    ax.set_xlim(0, 15)
    ax.set_ylim(0.5, len(tasks) + 0.5)
    ax.set_xticks(range(0, 16))
    ax.grid(True, axis='x', alpha=0.3, linestyle='--')
    
    legend_elements = [mpatches.Patch(facecolor=c, label=p) for p, c in phase_colors.items()]
    ax.legend(handles=legend_elements, loc='upper right', fontsize=11)
    
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight')
    plt.close()
    print(f"输出: {output_path}")

if __name__ == '__main__':
    sample_tasks = [
```
