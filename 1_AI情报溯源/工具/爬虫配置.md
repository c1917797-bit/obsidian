  - name: "New Model Release"
    trigger: "regex:GPT|Gemini|Claude|Llama|DeepSeek"
    channels: ["notification", "daily-report"]
    priority: P0

  - name: "Security Incident"
    trigger: "regex:vulnerability|exploit|attack|breach"
    channels: ["notification", "immediate-report"]
    priority: P0

  # 延迟告警 - P1事件

    trigger: "condition:citation_count > 100"
    channels: ["weekly-report"]
    priority: P1

  - name: "Major Funding"
    trigger: "regex:raised|Series|acquired|acquisition"
    amount_threshold: 100000000  # $100M
